วิธีติดตั้งสคริปต์บน Google Sheets (ครั้งแรก):

  1. เปิด Google Sheets ใหม่ -> เลือกเมนู ส่วนขยาย (Extensions) -> Apps Script
  2. ลบโค้ดเดิมใน Code.gs ออกทั้งหมด นำโค้ดด้านล่างไปวางแทนที่ แล้วกดบันทึก (Ctrl+S)
  3. กดปุ่ม การทำให้ใช้งานได้ (Deploy) -> การทำให้ใช้งานได้ใหม่ (New deployment)
  4. เลือกประเภท: Web app | ผู้มีสิทธิ์เข้าถึง: ทุกคน (Anyone) -> กด ทำให้ใช้งานได้ (Deploy)
  5. คัดลอก **Web app URL** นำมาวางในช่องด้านบน แล้วกดบันทึก URL
function doGet(e) {
  return handleRequest(e, "GET");
}

function doPost(e) {
  return handleRequest(e, "POST");
}

function handleRequest(e, method) {
  var lock = LockService.getScriptLock();
  try { lock.waitLock(5000); } catch (err) {}

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var action = "";
    var payloadData = null;

    if (e && e.parameter && e.parameter.action) {
      action = e.parameter.action;
    }

    if (e && e.postData && e.postData.contents) {
      try {
        payloadData = JSON.parse(e.postData.contents);
        if (payloadData && payloadData.action) action = payloadData.action;
      } catch (pErr) {}
    } else if (e && e.parameter && e.parameter.data) {
      try {
        payloadData = JSON.parse(e.parameter.data);
        if (payloadData && payloadData.action) action = payloadData.action;
      } catch (pErr) {}
    }

    // 1. ACTION: SYNC ALL (ผสานและบันทึกข้อมูลแบบป้องกันข้อมูลหาย 100%)
    if (action === "syncAll" && payloadData) {
      if (Array.isArray(payloadData.workers) && payloadData.workers.length > 0) {
        upsertToSheet(ss, "Workers", payloadData.workers, "id");
      }
      if (Array.isArray(payloadData.activities) && payloadData.activities.length > 0) {
        upsertToSheet(ss, "Activities", payloadData.activities, "id");
      }
      if (Array.isArray(payloadData.registrations)) {
        upsertRegistrations(ss, payloadData.registrations);
      }
      if (Array.isArray(payloadData.categories) && payloadData.categories.length > 0) {
        upsertCategories(ss, payloadData.categories);
      }
      if (Array.isArray(payloadData.admins) && payloadData.admins.length > 0) {
        upsertToSheet(ss, "Admins", payloadData.admins, "username");
      }

      var currentActivities = readFromSheet(ss, "Activities");
      var currentRegistrations = readFromSheet(ss, "Registrations");
      if (currentActivities.length > 0) {
        syncIndividualActivitySheets(ss, currentActivities, currentRegistrations);
      }

      SpreadsheetApp.flush(); // บันทึกข้อมูลลงชีททันทีไม่ต้องรอ
      saveBackupToGoogleDrive(payloadData); // สำรองข้อมูล JSON ลง Google Drive อัตโนมัติ
      return jsonResponse({ status: "success", message: "ผสานและบันทึกข้อมูลเรียบร้อยแล้ว" });
    }

    // 2. ACTION: GET ALL (อ่านข้อมูลทั้งหมด)
    var registrations = readAllRegistrationsCombined(ss);

    var result = {
      status: "success",
      workers: readFromSheet(ss, "Workers"),
      activities: readFromSheet(ss, "Activities"),
      registrations: registrations,
      categories: readFromSheet(ss, "Categories"),
      admins: readFromSheet(ss, "Admins")
    };

    return jsonResponse(result);
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  } finally {
    try { lock.releaseLock(); } catch (lErr) {}
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function syncIndividualActivitySheets(ss, activities, registrations) {
  if (!activities || !Array.isArray(activities)) return;
  var regs = Array.isArray(registrations) ? registrations : [];

  var regByAct = {};
  regs.forEach(function(r) {
    if (!r || !r.activityId) return;
    var actId = String(r.activityId).trim();
    if (!regByAct[actId]) regByAct[actId] = [];
    regByAct[actId].push(r);
  });

  activities.forEach(function(act) {
    if (!act || !act.id) return;
    var actId = String(act.id).trim();
    var safeTitle = (act.title || actId).replace(/[:\/\\?\*\[\]]/g, "_").substring(0, 15);
    var sheetName = "Act_" + actId + "_" + safeTitle;

    var actRegs = regByAct[actId] || [];
    writeToSheet(ss, sheetName, actRegs);
  });
}

function readRegistrationsFromActivitySheets(ss) {
  var sheets = ss.getSheets();
  var allRegs = [];
  var seenKeys = {};

  sheets.forEach(function(sheet) {
    var sName = sheet.getName();
    if (sName.indexOf("Act_") === 0) {
      var regs = readFromSheet(ss, sName);
      if (Array.isArray(regs)) {
        regs.forEach(function(r) {
          if (r && r.workerId && r.activityId) {
            var key = String(r.workerId).trim() + "_" + String(r.activityId).trim();
            if (!seenKeys[key]) {
              seenKeys[key] = true;
              allRegs.push(r);
            }
          }
        });
      }
    }
  });

  return allRegs;
}

function readAllRegistrationsCombined(ss) {
  var mainRegs = readFromSheet(ss, "Registrations");
  var actRegs = readRegistrationsFromActivitySheets(ss);
  var combinedMap = {};
  
  if (Array.isArray(mainRegs)) {
    mainRegs.forEach(function(r) {
      if (r && r.workerId && r.activityId) {
        var key = String(r.workerId).trim() + "_" + String(r.activityId).trim();
        combinedMap[key] = r;
      }
    });
  }
  
  if (Array.isArray(actRegs)) {
    actRegs.forEach(function(r) {
      if (r && r.workerId && r.activityId) {
        var key = String(r.workerId).trim() + "_" + String(r.activityId).trim();
        combinedMap[key] = r;
      }
    });
  }

  var result = [];
  Object.keys(combinedMap).forEach(function(k) {
    result.push(combinedMap[k]);
  });
  return result;
}

function upsertToSheet(ss, sheetName, incomingItems, keyField) {
  if (!incomingItems || !Array.isArray(incomingItems) || incomingItems.length === 0) return;
  var existing = readFromSheet(ss, sheetName);
  var mergedMap = {};

  existing.forEach(function(item) {
    if (item && item[keyField] !== undefined && item[keyField] !== null) {
      var k = String(item[keyField]).trim().toLowerCase();
      if (k) mergedMap[k] = item;
    }
  });

  incomingItems.forEach(function(item) {
    if (item && item[keyField] !== undefined && item[keyField] !== null) {
      var k = String(item[keyField]).trim().toLowerCase();
      if (k) {
        if (mergedMap[k]) {
          mergedMap[k] = Object.assign({}, mergedMap[k], item);
        } else {
          mergedMap[k] = item;
        }
      }
    }
  });

  var finalItems = [];
  Object.keys(mergedMap).forEach(function(k) { finalItems.push(mergedMap[k]); });
  writeToSheet(ss, sheetName, finalItems);
}

function upsertRegistrations(ss, incomingRegs) {
  if (!Array.isArray(incomingRegs)) return;
  var existing = readFromSheet(ss, "Registrations");
  var mergedMap = {};

  existing.forEach(function(r) {
    if (r && r.workerId && r.activityId && r.status !== 'cancelled') {
      var k = String(r.workerId).trim().toLowerCase() + "_" + String(r.activityId).trim().toLowerCase();
      mergedMap[k] = r;
    }
  });

  incomingRegs.forEach(function(r) {
    if (r && r.workerId && r.activityId) {
      var k = String(r.workerId).trim().toLowerCase() + "_" + String(r.activityId).trim().toLowerCase();
      if (r.status === 'cancelled') {
        delete mergedMap[k];
      } else if (mergedMap[k]) {
        mergedMap[k] = Object.assign({}, mergedMap[k], r);
      } else {
        mergedMap[k] = r;
      }
    }
  });

  var finalItems = [];
  Object.keys(mergedMap).forEach(function(k) { finalItems.push(mergedMap[k]); });
  writeToSheet(ss, "Registrations", finalItems);
}

function upsertCategories(ss, incomingCats) {
  if (!Array.isArray(incomingCats)) return;
  var existing = readFromSheet(ss, "Categories");
  var catSet = {};

  existing.forEach(function(c) {
    var str = String(c || "").trim();
    if (str) catSet[str] = true;
  });

  incomingCats.forEach(function(c) {
    var str = String(c || "").trim();
    if (str) catSet[str] = true;
  });

  var finalCats = Object.keys(catSet);
  writeToSheet(ss, "Categories", finalCats);
}

function writeToSheet(ss, sheetName, items) {
  if (!items || !Array.isArray(items) || items.length === 0) return;
  var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
  sheet.clearContents();
  
  if (typeof items[0] !== "object" || items[0] === null) {
    var catRows = [["Category"]].concat(items.map(function(item) { return [String(item)]; }));
    sheet.getRange(1, 1, catRows.length, 1).setValues(catRows);
    return;
  }

  var headerMap = {};
  items.forEach(function(item) {
    if (item && typeof item === "object") {
      Object.keys(item).forEach(function(k) { headerMap[k] = true; });
    }
  });
  var headers = Object.keys(headerMap);
  if (headers.length === 0) return;

  var allRows = [headers];
  items.forEach(function(item) {
    var row = headers.map(function(h) {
      var val = item[h];
      if (val === undefined || val === null) return "";
      if (typeof val === "object") return JSON.stringify(val);
      return String(val);
    });
    allRows.push(row);
  });

  sheet.getRange(1, 1, allRows.length, headers.length).setValues(allRows);
}

function readFromSheet(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  if (lastRow < 2 || lastCol < 1) return [];
  var data = sheet.getRange(1, 1, lastRow, lastCol).getDisplayValues();
  var headers = data[0];

  if (headers.length === 1 && headers[0] === "Category") {
    var catList = [];
    for (var i = 1; i < data.length; i++) {
      var cellVal = String(data[i][0] || "").trim();
      if (cellVal) catList.push(cellVal);
    }
    return catList;
  }

  var result = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    var hasContent = false;
    for (var j = 0; j < headers.length; j++) {
      var key = String(headers[j] || "").trim();
      if (!key) continue;
      var val = data[i][j];
      if (val === "" || val === null || val === undefined) {
        obj[key] = "";
      } else if (typeof val === "string" && (val.startsWith("{") || val.startsWith("["))) {
        try { obj[key] = JSON.parse(val); } catch(e) { obj[key] = val; }
      } else {
        obj[key] = val;
      }
      if (val !== "") hasContent = true;
    }
    if (hasContent) result.push(obj);
  }
  return result;
}

function saveBackupToGoogleDrive(payloadData) {
  if (!payloadData || typeof payloadData !== "object") return;
  try {
    var folderName = "SMO_Staff_Web_Backups";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);

    var formattedJson = JSON.stringify(payloadData, null, 2);
    var dateStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");

    // 1. สำรองไฟล์เวอร์ชันล่าสุด (Latest Backup)
    var latestFileName = "SMO_STAFF_BACKUP_LATEST.json";
    var latestFiles = folder.getFilesByName(latestFileName);
    if (latestFiles.hasNext()) {
      latestFiles.next().setContent(formattedJson);
    } else {
      folder.createFile(latestFileName, formattedJson, MimeType.PLAIN_TEXT);
    }

    // 2. สำรองไฟล์ประวัติประจำวัน (Daily Backup)
    var dailyFileName = "SMO_STAFF_BACKUP_" + dateStr + ".json";
    var dailyFiles = folder.getFilesByName(dailyFileName);
    if (dailyFiles.hasNext()) {
      dailyFiles.next().setContent(formattedJson);
    } else {
      folder.createFile(dailyFileName, formattedJson, MimeType.PLAIN_TEXT);
    }
  } catch (err) {
    Logger.log("Google Drive Backup Notice: " + err.toString());
  }
}
